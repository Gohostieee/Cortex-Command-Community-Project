cmake_minimum_required(VERSION 3.20)
project(CCRoomService LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
# Keep the standalone service on exactly the same bundled RakNet sources as the game.
file(READ "${REPO}/external/sources/RakNet/src/meson.build" RAKNET_MANIFEST)
string(REGEX MATCHALL "'[A-Za-z0-9_]+\\.cpp'" RAKNET_FILES "${RAKNET_MANIFEST}")
foreach(FILE IN LISTS RAKNET_FILES)
 string(REPLACE "'" "" FILE "${FILE}")
 list(APPEND RAKNET_SOURCES "${REPO}/external/sources/RakNet/src/${FILE}")
endforeach()
add_library(raknet STATIC ${RAKNET_SOURCES})
target_include_directories(raknet PUBLIC "${REPO}/external/sources/RakNet/include")
target_link_libraries(raknet PUBLIC Threads::Threads)
if(MSVC)
 target_compile_definitions(raknet PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
 target_compile_definitions(raknet PUBLIC NOMINMAX)
 target_link_libraries(raknet PUBLIC ws2_32 winmm)
else()
 target_compile_options(raknet PRIVATE -w -fpermissive)
endif()
add_executable(cc-room-service RoomService.cpp)
target_include_directories(cc-room-service PRIVATE "${REPO}/Source/System")
target_link_libraries(cc-room-service PRIVATE raknet)
add_executable(relay-tests "${REPO}/Tests/MultiplayerRelayTests.cpp" "${REPO}/Source/System/MultiplayerTransport.cpp")
target_include_directories(relay-tests PRIVATE "${REPO}/Source/System")
target_link_libraries(relay-tests PRIVATE raknet)
install(TARGETS cc-room-service RUNTIME DESTINATION bin)
